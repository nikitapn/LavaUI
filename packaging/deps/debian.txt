# Debian / Ubuntu packages for LavaUI + compositor + NPRPC.
# Lines starting with # are comments. A leading ? marks optional:
# install if the distro has it, otherwise the script builds wlroots 0.19.
#
# Tested against Debian 13 (trixie), Debian sid, and Ubuntu 24.04.
# Trixie and Noble ship wlroots 0.17/0.18 — 0.19 is built from source.

# ── build ──────────────────────────────────────────────────────────────────
build-essential
# Ubuntu 24.04's default g++ is 13; nprpc needs 14 (C++23 `this Self&`).
g++-14
gcc-14
ca-certificates
cmake
curl
git
meson
ninja-build
pkg-config
python3
rsync
tar
unzip
wget
xz-utils

libdbusmenu-glib-dev

libfreetype-dev

libglib2.0-dev

libharfbuzz-dev

libvulkan-dev
libx11-dev
# glfw3native.h includes Xrandr.h when GLFW_EXPOSE_NATIVE_X11 is set.
libxrandr-dev
libboost-stacktrace-dev
